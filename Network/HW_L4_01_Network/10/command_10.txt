sudo nano /etc/netplan/01-netcfg.yaml

# add these lines to the file

network:
  version: 2
  ethernets:
    eth0:
      dhcp4: no
      addresses:
        - 192.168.1.100/24
      gateway4: 192.168.1.1
      nameservers:
        addresses:
          - 1.1.1.1

# run command
sudo netplan apply

# test DNS
ping app.local
nslookup google.com
